-- Compare small gap layouts with direct byte-copy and list-edit calculations.
def checkGapEdges : IO Unit := do
  let samples : List (List UInt8) := [[], [0], [65], [255], [10, 65], [65, 0, 255], [65, 66, 67, 68]]
  let mut states := 0
  for text in samples do
    for split in List.range (text.length + 1) do
      for room in List.range 5 do
        let g := Gap02.gapAt text split room
        let l := Gap02.moveLeft g
        let r := Gap02.moveRight g
        let leftMemory := if split = 0 then g.memory else g.memory.set (g.finish - 1) (text[split - 1]?.getD 0)
        let rightMemory := if split = text.length then g.memory else g.memory.set g.start (text[split]?.getD 0)
        unless l.memory == leftMemory && r.memory == rightMemory do
          throw (IO.userError "movement did not match a single physical byte copy")
        unless l.text == text && r.text == text && l.capacity == g.capacity && r.capacity == g.capacity do
          throw (IO.userError "movement changed text or capacity")
        unless l.start == split - 1 && r.start == min (split + 1) text.length do
          throw (IO.userError "movement reached the wrong gap position")
        for point in List.range (text.length + 2) do
          unless g.memory[g.address point]? == text[point]? do
            throw (IO.userError "wrong physical read")
        match Gap02.insertByte g 88 with
        | none => unless room == 0 do throw (IO.userError "insertion wrongly refused")
        | some inserted =>
          unless room > 0 && inserted.text == text.take split ++ [88] ++ text.drop split &&
              inserted.capacity == g.capacity && inserted.free.length + 1 == room do
            throw (IO.userError "wrong insertion")
        let deleted := Gap02.deleteRight g
        unless deleted.text == text.take split ++ text.drop (split + 1) && deleted.memory == g.memory do
          throw (IO.userError "wrong deletion")
        for extra in [0, 1, 3] do
          let grown := Gap02.grow g extra
          unless grown.text == text && grown.capacity == g.capacity + extra do
            throw (IO.userError "wrong growth")
        states := states + 1
  IO.println s!"PASS {states} gap layouts, including empty text, zero free cells and both edges."

-- Run the boundary checks through Lean's interpreter.
#eval checkGapEdges
